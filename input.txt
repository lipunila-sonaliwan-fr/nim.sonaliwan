# a
a!
# akesi
tomo tawa akesi mi li wawa.
akesi li lon poka mi.
# ala
jan ale ala li pona.
moku suwi ala li lon insa poki.
soweli li moku ala.
ala li pakala.
# alasa
soweli alasa li tawa wawa.
alasa li ijo ike tawa mi.
mama mije mi li alasa e kala.
# ale
kasi ale li kama suli.
tenpo suno ale la ona li awen lon telo suli.
suno li suno lon tenpo ale.
ale li lon.
# anpa
jan anpa li pona.
ona li tawa anpa.
anpa li jaki.
jan li anpa lon sinpin pi jan sewi.
jan utala li anpa e akesi seli.
# ante
meli mi en jan ante li kama lon tomo lipu.
ante li pona.
jan sona li ante e ko ni.
# anu
ona anu jan ante li moku e kili.
sina pu anu seme?
# awen
tomo awen li jo e mani.
awen mi li pona.
ona li awen.
mije li awen e kasi.
jan lili li awen kute e kalama musi.
# e
mi moku e kili.
mi kute e waso.
# en
jan lili en mama ona li toki.
meli en mije en tonsi li sama.
# epiku
nasin epiku pi jan Ekule li pana e wawa tawa ona.
jan pi ma tomo mi li utala epiku e seli kepeken telo pi lupa telo ni.
# esun
jan esun li wile e mani mute.
esun li open.
jan lili li esun e moku suwi.
# ijo
jo ijo mute li ike e jan.
ona li lukin ijo e pali ona.
ijo ni li nasa.
jan sona li ijo e toki ona.
# ike
jan ike li utala e mi mute.
ona li pali ike e ko.
ike li lon.
akesi ni li ike.
pipi li ike e kasi mi.
# ilo
jan ilo li pali mute.
tomo lipu li jo e ilo nanpa.
# insa
kalama lili insa li lawa e mi mute.
moku li lon insa mi.
# jaki
mi moku ala e telo jaki.
ona li moku jaki.
o weka e jaki tan len sina.
jan ike li jaki e ma ona.
# jan
selo jan ni li jo e sitelen.
meli li toki jan tawa jan lili.
jan pi pona lukin li pali.
ona li jan e kiwen.
# jasima
sitelen pi lipu jasima li pona lukin.
jan utala tu li awen e lupa li tawa jasima.
jan pona mi la meli jasima ona li olin e mi mute.
supa pi tomo moku li jasima e suno pi tomo lupa.
# jelo
suno li jelo.
jelo li kule.
# jo
tomo ni li tomo jo.
jan utala li awen e jo mi.
jan ni li jo e lipu mute.
# kala
mi tawa lon telo kala.
mama mi li moku e kala lete.
# kalama
soweli kalama li kama.
o kute e kalama musi.
soweli li kalama.
meli sama mi li kalama e poki suli.
# kama
tenpo kama la mi lape.
ma tomo ni li jo ala e kama musi.
mi kama.
mi kama e pakala.
mi kama sona e toki pona.
# kasi
jan lili li musi lon ma kasi.
kasi li suli.
ni li kasi.
mije li kasi e ma ni.
# ken
sina sona e nasin ken anu seme ?
jan lawa li jo e ken.
ona li ken.
jan lawa li ken ala e ni: jan li moli e jan.
sina mute li ken moku e kili ni.
# kepeken
mi sitelen kepeken toki pona.
kepeken ni li ike.
sina kepeken e ilo moku.
# kijetesantakalu
kijetesantakalu li moku e pipi e kili.
# kili
jan lili li pali e telo kili.
ona li pali kili.
mi mute li moku e kili.
# kin
mi kin a!
kin la ona mute li pona.
# kipisi
jan lili li pana sama e kipisi pan tawa ona.
jan pali li kipisi e kasi suli.
# kiwen
ona li pali e seli kepeken ilo kiwen.
meli li pona kiwen e tomo ona.
kiwen li seli.
lete li kiwen e telo.
# ko
o pali e ko kepeken telo.
mije wawa li ko e kiwen suli.
# kokosila
jan kokosila li nasa.
o kokosila ala.
# kon
ilo pi tawa kon li kalama.
waso li tawa kon.
kon wawa li tawa e tomo tawa telo.
mi kon.
mi kon e ko walo.
# ku
sitelen ku li pana e sona.
ku li pana e pona tawa mi.
mi ku.
mi ku e toki mi.
# kule
len kule li pona tawa mi.
kule ni li suno.
jan lili li kule e sitelen mi.
# kulupu
ma kasi kulupu li lon poka pi tomo ni.
kulupu pi toki pona li wawa.
jan lawa li kulupu e jan utala.
# kute
ilo kute li wawa e kute.
kute pi meli ni li pona.
jan lili li kute e mama ona.
# la
open la ala li lon.
ken la tenpo pimeja li kama.
tenpo suno ni la jan pona mi li kama.
# lanpan
mani lanpan li pona e jan ala.
lanpan li ike a!
jan lawa li lanpan e ijo tan jan pali.
# lape
jan lili li tawa tomo lape ona.
jan pali ni li pali lape.
lape li pana e pona tawa sijelo mi.
meli mi li lape.
jan ike li lape e jan esun.
# laso
len noka laso sina li jaki.
laso li kule.
# lawa
jan lawa li toki e ijo suli.
ona li pilin lawa e ike.
lawa mi li jo ala e linja lawa.
meli li lawa e mije ona.
# leko
tomo mi li leko.
mama mije li pana e leko pi jan lili mute ona tawa poki.
# len
tenpo mute la meli li jo e poki len.
len ala la mi pilin lete.
meli li len e pan seli.
# lete
mi seli e moku soweli lete.
jan pi ma tomo li toki lete.
suno li weka e lete.
telo sin li lete e sijelo mi.
# li
ona li kute e waso.
ona li kama jo e len noka li tawa weka.
# lili
jan lili li musi.
mije mi li moku lili.
tenpo ale la jan li ken ala jo e tenpo pi jan lili.
jan alasa li lili e linja.
# linja
tenpo ni la tomo tawa linja li kama.
linja pi ilo toki li pakala.
# lipu
ma tomo mi li jo e tomo lipu.
mi kama sona tan lipu.
# loje
mi pana e telo loje mi.
meli li kule e len ona kepeken loje walo.
# lon
sina lon seme?
toki lon li wawa.
ken la kon sewi li pali e lon.
mi pilin la mi lon.
meli li lon e jan lili.
# luka
mi wile e ijo luka la mi wile e mani ala.
wan. tu, tu wan, tu tu, luka!
sijelo mi li jo e luka tu.
o luka e mi.
# lukin
ma Kanse li pona lukin.
mi lukin jo e tomo tawa.
nena ni li pana e lukin epiku.
mije li lukin wawa e meli.
# lupa

jan ike li lupa e sinpin tomo.
# ma

mi lon ma Inli.
# mama
mi tawa ma mama mi.
mama mi li toki wawa.
tenpo mute la meli li mama tawa jan ante.
# mani
nasin mani li ike.
tenpo kama lili la ona li moli mani.
mani li lawa e ale.
# meli
mama meli mi li pali e moku.
meli mute li wile ala kute e jan lawa.
# meso
jan utala la ma ni li meso.
jan sona li toki e ni: jan lili li pali meso.
jan sona li pali e meso ale.
# mi
ijo mi li lon tomo mi.
mi alasa e kala.
mi telo e mi.
# mije
mama mije mi li pali e moku.
mije ike li nasa.
# misikeke
ilo misikeke ni li ken kipisi.
open ale pi tenpo suno la mi moku e misikeke mi.
# moku
mi telo e ilo moku.
moku ni li suwi.
# moli
mi seli e sijelo pi soweli moli.
jan utala li utala moli.
jan sona li utala e moli.
mije li kama moli.
jan utala li moli e akesi seli.
# monsi

o awen lon monsi mi.
# monsuta

pimeja li monsuta e jan lili.
# mu

soweli li mu.
soweli li mu e ni: mi wile moku.
mu!

# musi
